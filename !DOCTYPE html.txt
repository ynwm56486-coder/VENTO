        function handleSaveDish(event) {
            event.preventDefault();
            const editId = document.getElementById('dish-edit-id').value;
            const name = document.getElementById('dish-name').value;
            const category = document.getElementById('dish-category').value;
            const price = parseFloat(document.getElementById('dish-price').value);
            const description = document.getElementById('dish-desc').value;
            const image = document.getElementById('dish-image').value || "https://images.unsplash.com/photo-1546069901-ba9599a7e63c?q=80&w=600&auto=format&fit=crop";

            const tags = [];
            if (document.getElementById('tag-vegan').checked) tags.push('vegan');
            if (document.getElementById('tag-vegetarian').checked) tags.push('vegetarian');
            if (document.getElementById('tag-gluten-free').checked) tags.push('gluten-free');
            if (document.getElementById('tag-spicy').checked) tags.push('spicy');

            if (editId) {
                // Update Existing
                const itemIndex = menuItems.findIndex(i => i.id === parseInt(editId));
                if (itemIndex > -1) {
                    menuItems[itemIndex] = { id: parseInt(editId), name, category, price, description, image, tags };
                    showToast("Dish updated successfully!");
                }
            } else {
                // Add New
                const newId = Date.now();
                menuItems.unshift({ id: newId, name, category, price, description, image, tags });
                showToast("New dish added to Vento menu!");
            }

            saveMenuData();
            renderMenu();
            closeDishEditorModal();
        }

                renderMenu();
                showToast("Item deleted from menu.");
            }
        }
